# PingDeck 多语言适配实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 PingDeck 接入简体中文、English、日本語三语言，覆盖界面与导出，并保持 PingCore 与展示资源解耦。

**Architecture:** String Catalog 放在 `PingDeck` target，SwiftPM 自动生成类型安全的 `LocalizedStringResource` 符号；PingCore 改为返回类型化错误与状态，导出通过 `ExportLabels` 注入文案。语言选择写入 `AppleLanguages`，重启后生效。

**Tech Stack:** Swift 6 / SwiftUI / SwiftPM / String Catalog（`.xcstrings`）/ XCTest

**Spec:** `docs/superpowers/specs/2026-11-06-i18n-design.md`

## Global Constraints

- 支持语言：`zh-Hans`、`en`、`ja`；不支持的系统语言回落 `en`。
- Catalog `sourceLanguage = zh-Hans`；`Package.swift` 的 `defaultLocalization = "en"`。
- 用户可见中文只能出现在 `.xcstrings` 与代码注释中；Swift 代码里不得再出现面向用户的中文字符串字面量。
- 生成的本地化符号是 `internal`，只允许在 `PingDeck` 模块使用；`PingCore` 不得引用。
- JSON 导出 schema 保持英文，不随语言变化。
- 每个任务结束必须 `swift build` 与 `swift test` 通过，并提交一个 Conventional Commit。
- 分支：`feat/i18n`；最低系统版本保持 macOS 14.4。
- 翻译清单是临时交付物，不提交仓库；`Localizable.xcstrings` 是唯一事实来源。

- [ ] **Step 3: 实现核心类型**

`Sources/PingCore/TargetValidator.swift`：

```swift
import Foundation

public enum TargetValidationIssue: Equatable, Sendable {
    case empty
    case protocolPrefix
    case path
    case whitespace
    case invalidCharacters
}

public enum TargetValidator {
    public static func validate(_ host: String) -> TargetValidationIssue? {
        let trimmed = host.trimmingCharacters(in: .whitespacesAndNewlines)
        if trimmed.isEmpty { return .empty }
        if trimmed.contains("://") { return .protocolPrefix }
        if trimmed.contains("/") { return .path }
        if trimmed.contains(" ") { return .whitespace }
        let allowed = CharacterSet(charactersIn: "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789.-_:[]%")
        if trimmed.rangeOfCharacter(from: allowed.inverted) != nil {
            return .invalidCharacters
        }
        return nil
    }
}
```

`Sources/PingCore/BulkImport.swift` 增加顶层枚举：

```swift
public enum IPv4RangeError: Equatable, Sendable {
    case ipv6Unsupported
    case malformedCIDR
    case cidrNotNetworkAddress
    case rangeTooLarge(limit: Int)
    case malformedRange
    case endBeforeStart
}
```

```swift
import XCTest

final class LocalizationCatalogTests: XCTestCase {
    private struct Catalog: Decodable {
        struct Entry: Decodable {
            struct Localization: Decodable {
                struct StringUnit: Decodable { let value: String }
                let stringUnit: StringUnit?
            }
            let comment: String?
            let localizations: [String: Localization]?
        }
        let sourceLanguage: String
        let strings: [String: Entry]
    }

    private let languages = ["en", "zh-Hans", "ja"]

    private func catalogURL() -> URL {
        URL(fileURLWithPath: #filePath)
            .deletingLastPathComponent()
            .deletingLastPathComponent()
            .deletingLastPathComponent()
            .appendingPathComponent("Sources/PingDeck/Resources/Localizable.xcstrings")
    }

    private func formatSpecifiers(in value: String) -> [String] {
        let pattern = #"%([0-9]+\$)?[-+0-9.l]*([@df])"#
        let regex = try! NSRegularExpression(pattern: pattern)
        let range = NSRange(value.startIndex..., in: value)
        return regex.matches(in: value, range: range).compactMap { match in
            guard let typeRange = Range(match.range(at: 2), in: value) else { return nil }
            return String(value[typeRange])
        }.sorted()
    }

    func testEveryKeyHasAllLanguagesAndMatchingPlaceholders() throws {
        let data = try Data(contentsOf: catalogURL())
        let catalog = try JSONDecoder().decode(Catalog.self, from: data)

        XCTAssertEqual(catalog.sourceLanguage, "zh-Hans")
        XCTAssertFalse(catalog.strings.isEmpty)

        for (key, entry) in catalog.strings {
            XCTAssertFalse(entry.comment?.isEmpty ?? true, "\(key) 缺少 comment")

            var values: [String] = []
            for language in languages {
                let value = entry.localizations?[language]?.stringUnit?.value
                XCTAssertFalse(value?.isEmpty ?? true, "\(key) 缺少 \(language) 文案")
                if let value { values.append(value) }
            }

            let expected = values.first.map(formatSpecifiers(in:)) ?? []
            for value in values.dropFirst() {
                XCTAssertEqual(
                    formatSpecifiers(in: value),
                    expected,
                    "\(key) 的格式占位符不一致：\(value)"
                )
            }
        }
    }
}
```
